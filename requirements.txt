# Requirements for DataanalysProjekt
# 
# Core dependencies for the Bayesian missing data models and evaluation pipeline
# (Custom Gibbs sampler in bayes_missing.py / bayes_missing_h.py - no PyMC required for those)
#
# To install in a controlled environment:
#   pip install -r requirements.txt
#
# Note: For the PyMC-based notebooks in bayesian_examples/ and bayesianUpdating/,
#       pymc>=5.0 is included. Some environments may need different versions.

# --- Numerical & scientific computing ---
numpy>=1.24,<2.0        # Core array operations, random number generation, linear algebra
scipy>=1.11,<1.15       # Special functions (expit/log_ndtr/ndtri), inverse Wishart sampling,
                        # numerical optimization used by custom Gibbs samplers

# --- Data manipulation ---
pandas>=2.0,<3.0        # DataFrame operations, CSV I/O, grouping for evaluation scripts

# --- Machine learning utilities ---
scikit-learn>=1.3,<1.7  # Used in evaluation/ for baselines, metrics (ROC/AUC), imputers
                        # (e.g., IterativeImputer in some comparison code)

# --- Visualization ---
matplotlib>=3.7,<3.11   # Plotting for diagnostics, recovery figures, and sensitivity plots

# --- Bayesian probabilistic programming (for PyMC notebooks) ---
pymc>=5.10,<6.0         # Used by bayesian_examples/ and some bayesianUpdating/ notebooks
                        # (NOT required by the pure-numpy Gibbs samplers in Preprocessing/ or evaluation/)
